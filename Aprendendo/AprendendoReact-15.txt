//Melhor maneira de fazer um formulário

import { useState } from "react"

function App() {

  const [name, setName] = useState("")

  function handleSubmit(e){
    e.preventDefault();

    console.log("Bem vindo", name)
  }

  return (
    <div>
      <h1>Form + Action</h1>

      <form onSubmit={handleSubmit}>
        <input
          type="text"
          value={name}
          onChange={(e) => setName(e.target.value)}
        />

        <button type="submit">Cadastrar</button>
      </form>
    </div>
  )
}

export default App


//Segundda maneira de fazer um formulário (Mais rápido e menos indicado)


import { useState } from "react";

function App() {

  const [message, setMessage] = useState("")

  //formData é chamado pelo action do form
  function handleRegister(formData){
    const nome = formData.get("nome")
    const tarefa = formData.get("tarefa")

    console.log(nome)
    console.log(tarefa)
    
    //Exibindo os componentes na tela
    setMessage("Bem vindo ," + nome + ", sua próxima tarefa será : "+ tarefa )
  }

  return (
    <div className="forms-container">
      <h1>Form + Action</h1>

      <form action={handleRegister}>
        <input
          type="text"
          name="nome"
          placeholder="Digite o seu nome"
          required
        />

        <input
          type="text"
          name="tarefa"
          placeholder="Digite qual será a sua proxima tarefa"
          required
        />

        <button type="submit">Cadastrar</button>
      </form>

      {/* Mensagem na tela  */}
      <h2>{message}</h2>
    </div>
  )
}

export default App
